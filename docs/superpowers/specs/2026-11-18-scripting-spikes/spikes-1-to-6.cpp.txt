static double Now() { return std::chrono::duration<double>(Clock::now().time_since_epoch()).count(); }

struct Mem { size_t used = 0, cap = SIZE_MAX, peak = 0; bool failed = false; };
static void* Alloc(void* ud, void* p, size_t o, size_t n)
{
    Mem* m = (Mem*)ud;
    if (n == 0) { if (p) { m->used -= o; free(p); } return nullptr; }
    size_t old = p ? o : 0;
    if (m->used - old + n > m->cap) { m->failed = true; return nullptr; }
    void* r = realloc(p, n);
    if (!r) return nullptr;
    m->used = m->used - old + n; if (m->used > m->peak) m->peak = m->used;
    return r;
}
static void OpenWhitelist(lua_State* L)
{
    luaopen_base(L); lua_pop(L,1);
    lua_pushcfunction(L, luaopen_string); lua_pushstring(L, LUA_STRLIBNAME); lua_call(L,1,0);
    lua_pushcfunction(L, luaopen_table);  lua_pushstring(L, LUA_TABLIBNAME); lua_call(L,1,0);
    lua_pushcfunction(L, luaopen_math);   lua_pushstring(L, LUA_MATHLIBNAME); lua_call(L,1,0);
    lua_pushcfunction(L, luaopen_bit);    lua_pushstring(L, LUA_BITLIBNAME); lua_call(L,1,0);
    lua_pushcfunction(L, luaopen_jit);    lua_pushstring(L, LUA_JITLIBNAME); lua_call(L,1,0);
    // strip dangerous
    const char* kill[] = {"loadfile","dofile","load","loadstring","collectgarbage","getfenv","setfenv","rawget_unused",nullptr};
    for (int i = 0; kill[i]; ++i) { lua_pushnil(L); lua_setglobal(L, kill[i]); }
    // hide jit from scripts
    lua_pushnil(L); lua_setglobal(L, "jit");
    lua_pushnil(L); lua_setglobal(L, "package");
    lua_pushnil(L); lua_setglobal(L, "require");
}
static lua_State* NewState(Mem* m, bool whitelist = true)
{
    lua_State* L = lua_newstate(Alloc, m);
    if (!L) return nullptr;
    if (whitelist) OpenWhitelist(L);
    return L;
}

    std::vector<float> px, py, pz, vx, vy, vz;      // live
    std::vector<float> spx, spy, spz;               // tick-start snapshot
    uint64_t Hash() const
    {
        uint64_t h = 1469598103934665603ull;
        auto mix = [&](float f){ uint32_t u; memcpy(&u,&f,4); h ^= u; h *= 1099511628211ull; };
        for (int i = 0; i < N; ++i) { mix(px[i]);mix(py[i]);mix(pz[i]);mix(vx[i]);mix(vy[i]);mix(vz[i]); }
        return h;

};

static void InitPart(Part& p, World* w)
{
    p.w = w; p.L = NewState(&p.mem);
    lua_State* L = p.L;
    lua_pushlightuserdata(L, &p); lua_pushcclosure(L, b_pos, 1); lua_setglobal(L, "pos");
    lua_pushlightuserdata(L, &p); lua_pushcclosure(L, b_setvel, 1); lua_setglobal(L, "setvel");
    (void)b_rand;
    if (luaL_loadbuffer(L, kScript, strlen(kScript), "=s") || lua_pcall(L,0,0,0)) { printf("load err %s\n", lua_tostring(L,-1)); exit(1); }
    lua_getglobal(L, "math"); lua_getfield(L, -1, "randomseed"); lua_pushnumber(L, 12345 + (double)(&p == nullptr)); lua_pcall(L,1,0,0); lua_pop(L,1);
}

// simple fixed pool: run fn(i) for i in [0,count) across T threads (dynamic claim)
static void ParFor(int T, int count, const std::function<void(int)>& fn)
{
    if (T <= 1) { for (int i=0;i<count;++i) fn(i); return; }
    std::atomic<int> next{0};
    std::vector<std::thread> th;
    for (int t=0;t<T;++t) th.emplace_back([&]{ for(;;){ int i=next.fetch_add(1); if(i>=count) break; fn(i);} });
    for (auto& t: th) t.join();
}

static uint64_t RunTicks(int N, int P, int T, int ticks, double* msPerTick, double* mergeMs)
{
    World w; w.N=N; for (auto* v : {&w.px,&w.py,&w.pz,&w.vx,&w.vy,&w.vz,&w.spx,&w.spy,&w.spz}) v->assign(N,0.f);
    for (int i=0;i<N;++i){ w.px[i]=(float)(i%100); w.py[i]=(float)((i/100)%100); w.pz[i]=(float)(i/10000); }
    std::vector<Part> parts(P);
    for (int i=0;i<N;++i) parts[(uint32_t)(i*2654435761u>>7)%P].ents.push_back(i);
    for (auto& p : parts) InitPart(p, &w);
    double t0=Now(), merge=0;
    for (int t=0;t<ticks;++t)
    {
        w.spx=w.px; w.spy=w.py; w.spz=w.pz;
        ParFor(T, P, [&](int pi){
            Part& p=parts[pi]; p.cmds.clear(); p.seq=0; lua_State* L=p.L;
            for (int e : p.ents) { lua_getglobal(L,"on_update"); lua_pushinteger(L,e); lua_pushinteger(L,(e+1)%N); lua_pushnumber(L,1.0/60); if (lua_pcall(L,3,0,0)) { printf("err %s\n", lua_tostring(L,-1)); exit(1);} }
        });
        double m0=Now();
        for (auto& p : parts) for (auto& c : p.cmds) { w.vx[c.entity]=c.x; w.vy[c.entity]=c.y; w.vz[c.entity]=c.z; }
        for (int i=0;i<N;++i){ w.px[i]+=w.vx[i]/60; w.py[i]+=w.vy[i]/60; w.pz[i]+=w.vz[i]/60; }
        merge += Now()-m0;
    }
    *msPerTick=(Now()-t0)*1000/ticks; *mergeMs=merge*1000/ticks;
    uint64_t h=w.Hash();
    for (auto& p : parts) lua_close(p.L);
    return h;
}

struct Budget { bool tripped=false; std::atomic<long> count{0}; long limit; double deadline; };
static thread_local Budget* gB;
static void Hook(lua_State* L, lua_Debug*) { if (gB) { gB->count += 1000; if (gB->count > gB->limit || Now()>gB->deadline) { gB->tripped=true; luaL_error(L, "script budget exceeded"); } } }
static int SafePcall(lua_State* L) { int n=lua_gettop(L); luaL_checkany(L,1); int rc=lua_pcall(L,n-1,LUA_MULTRET,0); if (gB && gB->tripped) return lua_error(L); lua_pushboolean(L,rc==0); lua_insert(L,1); return lua_gettop(L); }
static int SafeResume(lua_State* L) { lua_State* co=lua_tothread(L,1); int n=lua_gettop(L)-1; lua_xmove(L,co,n); int rc=lua_resume(co,n); if (gB && gB->tripped) { lua_pushstring(L,"script budget exceeded"); return lua_error(L);} if(rc==0||rc==LUA_YIELD){ int m=lua_gettop(co); lua_xmove(co,L,m); lua_pushboolean(L,1); lua_insert(L,-m-1); return m+1;} lua_pushboolean(L,0); lua_xmove(co,L,1); return 2; }
