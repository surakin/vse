// Spike 1a: libasound, device "default" (pipewire-alsa) and "hw:3,0"; period 256 and 1024 frames @48k stereo S16/F32; plays silence;
// reports granted params, snd_pcm_delay, and the wall time to write 200 periods (pacing jitter).
#include <alsa/asoundlib.h>
#include <time.h>
#include <stdio.h>
#include <vector>
#include <algorithm>
static double now(){timespec t;clock_gettime(CLOCK_MONOTONIC,&t);return t.tv_sec*1e3+t.tv_nsec/1e6;}
int main(int argc,char**argv){
  const char*dev=argv[1]; unsigned period=atoi(argv[2]);
  snd_pcm_t*pcm; int e=snd_pcm_open(&pcm,dev,SND_PCM_STREAM_PLAYBACK,0);
  if(e<0){printf("%s: open failed: %s\n",dev,snd_strerror(e));return 0;}
  snd_pcm_hw_params_t*hw; snd_pcm_hw_params_alloca(&hw); snd_pcm_hw_params_any(pcm,hw);
  snd_pcm_hw_params_set_access(pcm,hw,SND_PCM_ACCESS_RW_INTERLEAVED);
  const char* fmt="F32";
  if(snd_pcm_hw_params_set_format(pcm,hw,SND_PCM_FORMAT_FLOAT_LE)<0){snd_pcm_hw_params_set_format(pcm,hw,SND_PCM_FORMAT_S16_LE);fmt="S16";}
  snd_pcm_hw_params_set_channels(pcm,hw,2);
  unsigned rate=48000; snd_pcm_hw_params_set_rate_near(pcm,hw,&rate,0);
  snd_pcm_uframes_t p=period; snd_pcm_hw_params_set_period_size_near(pcm,hw,&p,0);
  snd_pcm_uframes_t buf=p*3; snd_pcm_hw_params_set_buffer_size_near(pcm,hw,&buf);
  e=snd_pcm_hw_params(pcm,hw); if(e<0){printf("%s: hw_params failed: %s\n",dev,snd_strerror(e));return 0;}
  printf("%s: fmt %s rate %u period %lu buffer %lu\n",dev,fmt,rate,(unsigned long)p,(unsigned long)buf);
  std::vector<char> data(p*2*4,0);
  std::vector<double> dt; double last=now(); long under=0; snd_pcm_sframes_t delay=0;
  for(int i=0;i<200;i++){
    snd_pcm_sframes_t n=snd_pcm_writei(pcm,data.data(),p);
    if(n<0){under++; snd_pcm_recover(pcm,(int)n,1);}
    double t=now(); dt.push_back(t-last); last=t;
    if(i==100) snd_pcm_delay(pcm,&delay);
  }
  std::sort(dt.begin()+20,dt.end()); double med=dt[110], mx=dt.back();
  printf("  median write interval %.3f ms (period %.3f ms), max %.3f ms, delay %ld frames (%.2f ms), xruns %ld\n",med,1000.0*p/rate,mx,(long)delay,1000.0*delay/rate,under);
  snd_pcm_close(pcm);
}
