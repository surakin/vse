# Phase 15 Day 194: the window adapter, the one library that links both ui and platform (ui itself names no window). One library per sanitizer flavor, like ui.
set(UI_WINDOW_SOURCES window_adapter.cpp)
set(UI_WINDOW_INCLUDE_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)

if(CMAKE_SYSTEM_NAME STREQUAL "Windows")
  set(_flavors "")
  if(ENGINE_HAVE_ASAN)
    list(APPEND _flavors "_asan")
  endif()
  foreach(suffix "" ${_flavors})
    add_library(ui_window${suffix} STATIC ${UI_WINDOW_SOURCES})
    target_include_directories(ui_window${suffix} PUBLIC ${UI_WINDOW_INCLUDE_ROOT})
    target_compile_options(ui_window${suffix} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:${ENGINE_PRODUCTION_COMPILE_OPTIONS}>)
    target_link_libraries(ui_window${suffix} PUBLIC ui${suffix} platform${suffix})
  endforeach()
  return()
endif()

foreach(suffix "" "_asan" "_tsan")
  add_library(ui_window${suffix} STATIC ${UI_WINDOW_SOURCES})
  target_include_directories(ui_window${suffix} PUBLIC ${UI_WINDOW_INCLUDE_ROOT})
  target_compile_options(ui_window${suffix} PRIVATE ${ENGINE_PRODUCTION_COMPILE_OPTIONS})
  target_link_libraries(ui_window${suffix} PUBLIC ui${suffix} platform${suffix})
endforeach()
