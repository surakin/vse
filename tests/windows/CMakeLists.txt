# Windows-only tests (Phase 13). Configured only on Windows (root CMakeLists.txt); labeled phase 13 through the `testopt:windows_*`
# line of tests/phases.txt (these tests do not exist on Linux).
#
#   windows_toolchain_discovery_msvc, _clang_cl  the toolchain files find MSVC, the Windows SDK and clang-cl from a stripped environment
#   windows_header_check     every foundation header, then os/windows/win32.h, then uses of the clashing names: compiles
#   windows_header_planted   the same with a bare <windows.h>: must fail, naming the renamed LoadLibrary (the wrapper is what prevents it)
#   windows_manifest         an engine-built executable runs with the manifest's UTF-8 code page and per-monitor-v2 DPI awareness
#   windows_platform_tests   the Win32 window (Day 158), its input (Day 159) and ENGINE_WINDOWED_PROGRAM (subsystem, arguments, output), with
#                            windows_windowed_probe
#   windows_docgen_tests     docgen's child processes on Windows (Day 161): arguments, output files, working directory, exit and exception
#                            status, killing a tree; with windows_child_probe

# --- The header check: generated from the real list of foundation headers, so a new header is covered without an edit here -------
file(GLOB_RECURSE _foundation_headers RELATIVE "${CMAKE_SOURCE_DIR}/engine/foundation" CONFIGURE_DEPENDS
     "${CMAKE_SOURCE_DIR}/engine/foundation/*.h")
set(_includes "")
foreach(_h IN LISTS _foundation_headers)
  # Per-platform headers, generated code, the unity macro pairs and the platform halves of neutral interfaces are not API a
  # Windows file would include next to <windows.h>.
  if(_h MATCHES "(^|/)(linux|macos|windows|generated|unity)/" OR _h MATCHES "_platform\\.h$" OR _h MATCHES "_linux\\.h$")
    continue()
  endif()
  string(APPEND _includes "#include \"${_h}\"\n")
endforeach()
foreach(_variant check planted)
  if(_variant STREQUAL "check")
    set(_windows "#include \"os/windows/win32.h\"")
  else()
    set(_windows "#include <windows.h> // bare, without unity/windows_enable.h and windows_disable.h")
  endif()
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/windows_header_${_variant}.cpp.in"
       "// Generated by tests/windows/CMakeLists.txt.\n${_includes}\n${_windows}\n\n#include \"${CMAKE_CURRENT_SOURCE_DIR}/header_uses.inc\"\n")
  configure_file("${CMAKE_CURRENT_BINARY_DIR}/windows_header_${_variant}.cpp.in" "${CMAKE_CURRENT_BINARY_DIR}/windows_header_${_variant}.cpp" COPYONLY)
  add_library(windows_header_${_variant} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/windows_header_${_variant}.cpp")
  target_include_directories(windows_header_${_variant} PRIVATE "${CMAKE_SOURCE_DIR}/engine/foundation"
                             "${ENGINE_VULKAN_HEADERS_DIR}")
  target_compile_options(windows_header_${_variant} PRIVATE ${ENGINE_PRODUCTION_COMPILE_OPTIONS})
  set_target_properties(windows_header_${_variant} PROPERTIES UNITY_BUILD OFF)
endforeach()
set_target_properties(windows_header_planted PROPERTIES EXCLUDE_FROM_ALL TRUE)

# Each check builds its target inside the build tree; one at a time, so two ninja runs never share the build directory.
add_test(NAME windows_header_check COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR} --target windows_header_check)
add_test(NAME windows_header_planted COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR} --target windows_header_planted)
# The planted build must fail for the reason the wrapper exists: `os::LoadLibrary` declared by the engine header, then renamed to
# LoadLibraryW by the macro at the use. Matching the message keeps a failure for another reason (a missing header) from passing.
set_tests_properties(windows_header_planted PROPERTIES PASS_REGULAR_EXPRESSION "LoadLibraryW.*is not a member|no member named .LoadLibraryW.")  # cl | clang-cl
set_tests_properties(windows_header_check windows_header_planted PROPERTIES RESOURCE_LOCK build_tree)

# --- The Windows os backend and the UTF-16 helpers (Day 149): Catch2 tests plus a child program they start -------------------------
find_package(Threads REQUIRED)
include(Catch)
add_executable(windows_os_probe os_probe.cpp)
target_link_libraries(windows_os_probe PRIVATE foundation_base)
target_compile_options(windows_os_probe PRIVATE ${ENGINE_PRODUCTION_COMPILE_OPTIONS})
function(engine_add_windows_os_tests target foundation_lib suffix label)
  add_executable(${target} os_windows_test.cpp)
  target_link_libraries(${target} PRIVATE ${foundation_lib} Catch2::Catch2WithMain Threads::Threads)
  target_include_directories(${target} PRIVATE "${CMAKE_SOURCE_DIR}/tests/support")
  target_compile_definitions(${target} PRIVATE "WINDOWS_OS_PROBE_PATH=L\"$<TARGET_FILE:windows_os_probe>\"")
  add_dependencies(${target} windows_os_probe)
  if(label STREQUAL "asan")
    engine_asan_runtime(${target})
  endif()
  # A probe that hangs (a pipe write that never returns) must fail the test, not stall the run.
  engine_discover_tests(${target} TEST_SUFFIX "${suffix}" PROPERTIES LABELS "${label}" TIMEOUT 60)
endfunction()
engine_add_windows_os_tests(windows_os_tests foundation_base "" "windows")
if(ENGINE_HAVE_ASAN)
  engine_add_windows_os_tests(windows_os_tests_asan foundation_base_asan " [asan]" "asan")
endif()

# --- The Win32 window and the windowed programs' entry point (Day 158): Catch2 tests plus a GUI program they start ------------------------
add_executable(windows_windowed_probe windowed_probe.cpp)
target_link_libraries(windows_windowed_probe PRIVATE platform)
target_compile_options(windows_windowed_probe PRIVATE ${ENGINE_PRODUCTION_COMPILE_OPTIONS})
function(engine_add_windows_platform_tests target platform_lib suffix label)
  add_executable(${target} win32_window_windows_test.cpp win32_input_windows_test.cpp win32_events_windows_test.cpp
                           win32_clipboard_windows_test.cpp win32_dragdrop_windows_test.cpp)
  target_link_libraries(${target} PRIVATE ${platform_lib} Catch2::Catch2WithMain Threads::Threads)
  target_include_directories(${target} PRIVATE "${CMAKE_SOURCE_DIR}/tests/support")
  target_compile_definitions(${target} PRIVATE "WINDOWED_PROBE_PATH=L\"$<TARGET_FILE:windows_windowed_probe>\""
                             "WINDOWED_PROBE_PATH_UTF8=\"$<TARGET_FILE:windows_windowed_probe>\"")
  add_dependencies(${target} windows_windowed_probe)
  if(label STREQUAL "asan")
    engine_asan_runtime(${target})
  endif()
  # The window cases share the desktop with the other window tests (the `display` resource, as in tests/foundation).
  engine_discover_tests(${target} TEST_SPEC "~[window]" TEST_SUFFIX "${suffix}" PROPERTIES LABELS "${label}" TIMEOUT 60)
  engine_discover_tests(${target} TEST_SPEC "[window]" TEST_SUFFIX "${suffix}" PROPERTIES LABELS "${label};display" RESOURCE_LOCK display
                        TIMEOUT 60)
endfunction()
engine_add_windows_platform_tests(windows_platform_tests platform "" "windows")
if(ENGINE_HAVE_ASAN)
  engine_add_windows_platform_tests(windows_platform_tests_asan platform_asan " [asan]" "asan")
endif()

# --- Toolchain discovery from a stripped environment -----------------------------------------------------------------------
foreach(_tc msvc clang-cl)
  string(REPLACE "-" "_" _name "${_tc}")
  add_test(NAME windows_toolchain_discovery_${_name}
           COMMAND ${CMAKE_COMMAND} -DTOOLCHAIN=${CMAKE_SOURCE_DIR}/cmake/toolchains/windows-${_tc}.cmake
                   -DWORK=${ENGINE_TEST_TMP_DIR}/windows_discovery-${_tc} -DPROJECT_SOURCE=${CMAKE_CURRENT_SOURCE_DIR}/discovery
                   -DNINJA=${CMAKE_MAKE_PROGRAM} -P ${CMAKE_CURRENT_SOURCE_DIR}/discovery/check.cmake)
  set_tests_properties(windows_toolchain_discovery_${_name} PROPERTIES PASS_REGULAR_EXPRESSION "discovery ok: hello from")
endforeach()
