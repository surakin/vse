# macOS-only tests (Phase 26, a side phase on the Mac). Configured only on macOS (root CMakeLists.txt); labeled phase 26 through
# tests/phases.txt (the Catch2 file by its path, the build checks by their `test:macos_*` line).
#
#   macos_build_defs_tests      build_defs.h on Apple clang (platform, processor, compiler, the sanitizer of each flavor), plain, asan, tsan
#   macos_os_tests              the os backend (os/macos/os.cpp, session.mm): clock, sleeps, pages, process figures, LoadLibrary, a child
#                               (macos_os_child) reading non-ASCII arguments and logging UTF-8 to a pipe; plain, asan, tsan
#   macos_header_apple_first    every neutral foundation header compiles as Objective-C++ (ARC, the warnings of every target) after
#                               <AppKit/AppKit.h>, <Metal/Metal.h> and <QuartzCore/CAMetalLayer.h>: no Apple macro or type breaks them
#   macos_header_apple_last     the same with the frameworks after the engine headers, the other order a unity build can merge them in
#   macos_production_options_planted  an engine library's .mm holding a `try`: must fail (the production options reach Objective-C++)
#   macos_header_planted        the frameworks with Apple's old AssertMacros.h names switched on, then a function named verify: must fail,
#                               which shows the check catches a clash and why ENGINE_MACOS_DEFINITIONS pins the setting to 0
# Day 299's Spike B found no clash, so no unity/apple_enable.h pair exists (CLAUDE.md "Unity builds": a pair answers a real conflict).

# --- The header checks: generated from the real list of foundation headers, so a new header is covered without an edit here -------
file(GLOB_RECURSE _foundation_headers RELATIVE "${CMAKE_SOURCE_DIR}/engine/foundation" CONFIGURE_DEPENDS
     "${CMAKE_SOURCE_DIR}/engine/foundation/*.h")
set(_includes "")
foreach(_h IN LISTS _foundation_headers)
  # Per-platform headers, generated code, the unity macro pairs and the platform halves of neutral interfaces are not what a macOS file
  # includes next to the frameworks; the GPU driver probe is not part of the macOS build (Phase 26 decision 8).
  if(_h MATCHES "(^|/)(linux|macos|windows|posix|generated|unity)/" OR _h MATCHES "_platform\\.h$" OR _h MATCHES "_linux\\.h$"
     OR _h STREQUAL "os/gpu_driver_probe.h")
    continue()
  endif()
  string(APPEND _includes "#include \"${_h}\"\n")
endforeach()
set(_frameworks "#import <AppKit/AppKit.h>\n#import <Metal/Metal.h>\n#import <QuartzCore/CAMetalLayer.h>\n")
set(_header_first "// Generated by tests/macos/CMakeLists.txt.\n${_frameworks}\n${_includes}")
set(_header_last "// Generated by tests/macos/CMakeLists.txt.\n${_includes}\n${_frameworks}")
set(_header_planted "// Generated by tests/macos/CMakeLists.txt.\n#undef __ASSERT_MACROS_DEFINE_VERSIONS_WITHOUT_UNDERSCORES\n#define __ASSERT_MACROS_DEFINE_VERSIONS_WITHOUT_UNDERSCORES 1\n${_frameworks}\n${_includes}\n#include \"${CMAKE_CURRENT_SOURCE_DIR}/header_uses_planted.inc\"\n")
foreach(_variant apple_first apple_last planted)
  file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/macos_header_${_variant}.mm" CONTENT "${_header_${_variant}}")
  add_library(macos_header_${_variant} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/macos_header_${_variant}.mm")
  target_include_directories(macos_header_${_variant} PRIVATE "${CMAKE_SOURCE_DIR}/engine/foundation")
  target_compile_options(macos_header_${_variant} PRIVATE ${ENGINE_PRODUCTION_COMPILE_OPTIONS})
  set_target_properties(macos_header_${_variant} PROPERTIES UNITY_BUILD OFF EXCLUDE_FROM_ALL TRUE)
  # Each check builds its target inside the build tree; one at a time, so two ninja runs never share the build directory.
  add_test(NAME macos_header_${_variant} COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR} --target macos_header_${_variant})
  set_tests_properties(macos_header_${_variant} PROPERTIES RESOURCE_LOCK build_tree)
endforeach()
# The planted build must fail for the reason the setting exists: AssertMacros.h's verify macro expanding inside the declaration.
# Matching the message keeps a failure for another reason (a missing header) from passing.
set_tests_properties(macos_header_planted PROPERTIES PASS_REGULAR_EXPRESSION "header_uses_planted\\.inc:[0-9]+:[0-9]+: error: .*")
engine_label_tests(macos_header_apple_first macos_header_apple_last macos_header_planted)

# --- build_defs.h on Apple clang, in the three flavors ----------------------------------------------------------------------------------
find_package(Threads REQUIRED)
include(Catch)
function(engine_add_macos_build_defs_tests target foundation_lib suffix label expect_asan expect_tsan)
  add_executable(${target} build_defs_macos_test.cpp)
  target_link_libraries(${target} PRIVATE ${foundation_lib} Catch2::Catch2WithMain Threads::Threads)
  target_compile_definitions(${target} PRIVATE ENGINE_TEST_EXPECT_ASAN=${expect_asan} ENGINE_TEST_EXPECT_TSAN=${expect_tsan})
  engine_discover_tests(${target} TEST_SUFFIX "${suffix}" PROPERTIES LABELS "${label}")
endfunction()
engine_add_macos_build_defs_tests(macos_build_defs_tests foundation_base "" "macos" 0 0)
engine_add_macos_build_defs_tests(macos_build_defs_tests_asan foundation_base_asan " [asan]" "asan" 1 0)
engine_add_macos_build_defs_tests(macos_build_defs_tests_tsan foundation_base_tsan " [tsan]" "tsan" 0 1)

# --- Objective-C++ in an engine library gets the production options (no exceptions, no RTTI) ------------------------------------------
# A .mm built by the same function as the foundation's libraries, holding a `try`: the build must fail on it.
engine_add_foundation_layer(macos_production_options_planted "" production_options_planted.mm)
set_target_properties(macos_production_options_planted PROPERTIES UNITY_BUILD OFF EXCLUDE_FROM_ALL TRUE)
add_test(NAME macos_production_options_planted
         COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR} --target macos_production_options_planted)
set_tests_properties(macos_production_options_planted PROPERTIES RESOURCE_LOCK build_tree
                     PASS_REGULAR_EXPRESSION "production_options_planted\\.mm:[0-9]+:[0-9]+: error: cannot use 'try' with exceptions disabled")
engine_label_tests(macos_production_options_planted)

# --- The os backend (Day 301): the clock across threads, sleeps, pages, process figures, library loading, a child's arguments and log -------
add_executable(macos_os_child os_child_macos.cpp)
target_link_libraries(macos_os_child PRIVATE foundation_base)
function(engine_add_macos_os_tests target foundation_lib suffix label)
  add_executable(${target} os_macos_test.cpp)
  target_link_libraries(${target} PRIVATE ${foundation_lib} Catch2::Catch2WithMain Threads::Threads)
  target_compile_definitions(${target} PRIVATE ENGINE_TEST_CHILD="$<TARGET_FILE:macos_os_child>")
  add_dependencies(${target} macos_os_child)
  engine_discover_tests(${target} TEST_SUFFIX "${suffix}" PROPERTIES LABELS "${label}")
endfunction()
engine_add_macos_os_tests(macos_os_tests foundation_base "" "macos")
engine_add_macos_os_tests(macos_os_tests_asan foundation_base_asan " [asan]" "asan")
engine_add_macos_os_tests(macos_os_tests_tsan foundation_base_tsan " [tsan]" "tsan")
