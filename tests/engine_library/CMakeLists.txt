# Day 343: a C program per flavor links the shared library (ENGINE_BUILD_SHARED, Linux) and runs jobs inside it. The sanitized libraries load only into a
# program built with the same sanitizer (spec, Spike 3). Without the option the three tests exist as skipped stubs, so that the phase has the same flavors in
# every configuration (test_phase_labels).
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  foreach(suffix "" "_asan" "_tsan")
    set(flavor_label "")
    if(suffix STREQUAL "_asan")
      set(flavor_label asan)
    elseif(suffix STREQUAL "_tsan")
      set(flavor_label tsan)
    endif()
    if(ENGINE_BUILD_SHARED AND TARGET engine${suffix})
      add_executable(shared_link_probe${suffix} shared_link_probe.c)
      target_include_directories(shared_link_probe${suffix} PRIVATE ${CMAKE_SOURCE_DIR}/engine)
      target_link_libraries(shared_link_probe${suffix} PRIVATE engine${suffix})
      set_target_properties(shared_link_probe${suffix} PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:engine${suffix}>")
      if(suffix STREQUAL "_asan")
        target_compile_options(shared_link_probe_asan PRIVATE ${ENGINE_ASAN_COMPILE_OPTIONS})
        target_link_options(shared_link_probe_asan PRIVATE ${ENGINE_ASAN_LINK_OPTIONS})
      elseif(suffix STREQUAL "_tsan")
        target_compile_options(shared_link_probe_tsan PRIVATE ${ENGINE_TSAN_COMPILE_OPTIONS})
        target_link_options(shared_link_probe_tsan PRIVATE ${ENGINE_TSAN_LINK_OPTIONS})
      endif()
      add_test(NAME engine_shared_link_probe${suffix} COMMAND shared_link_probe${suffix})
      # The exit code decides (the probe returns 1 for a wrong count): sanitizer reports and teardown crashes after the output must fail the test too.
    else()
      if(ENGINE_BUILD_SHARED)
        set(why "this compiler has no ${flavor_label} flavor, so no engine${suffix} library was built")
      else()
        set(why "ENGINE_BUILD_SHARED is OFF")
      endif()
      add_test(NAME engine_shared_link_probe${suffix} COMMAND ${CMAKE_COMMAND} -E echo "skipped: ${why}")
      set_tests_properties(engine_shared_link_probe${suffix} PROPERTIES SKIP_REGULAR_EXPRESSION "skipped: ")
    endif()
    set_tests_properties(engine_shared_link_probe${suffix} PROPERTIES LABELS "engine_library;${flavor_label}")
    engine_label_tests(engine_shared_link_probe${suffix})
  endforeach()
endif()

# Day 345 (one of everything across the boundary): C hosts and plugins that load the shared library with dlopen and check that the engine's
# singletons exist once, that nothing crosses the boundary to be freed by the other side, that the library unloads after its jobs ran, and that
# the crash handler comes and goes cleanly. They read the state table through the extension `engine.test_state` of the C API (Day 346).
# Without ENGINE_BUILD_SHARED every test exists as a skipped stub, so that the phase has the same tests in every configuration.
function(engine_library_c_test name suffix)
  # engine_library_c_test(<test> <flavor suffix> SOURCE <file> [DEFINES ...] [PLANTED] [PASS_REGEX <re>] [PLUGIN <target>])
  cmake_parse_arguments(ARG "PLANTED" "SOURCE;PASS_REGEX;PLUGIN" "DEFINES;ENV" ${ARGN})
  set(flavor_label "")
  if(suffix STREQUAL "_asan")
    set(flavor_label asan)
  elseif(suffix STREQUAL "_tsan")
    set(flavor_label tsan)
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND ENGINE_BUILD_SHARED AND TARGET engine${suffix})
    add_executable(${name} ${ARG_SOURCE})
    target_include_directories(${name} PRIVATE ${CMAKE_SOURCE_DIR}/engine)
    target_compile_definitions(${name} PRIVATE "ENGINE_LIB_PATH=\"$<TARGET_FILE:engine${suffix}>\"" ${ARG_DEFINES})
    if(ARG_PLUGIN)
      target_compile_definitions(${name} PRIVATE "PLUGIN_PATH=\"$<TARGET_FILE:${ARG_PLUGIN}>\"")
      add_dependencies(${name} ${ARG_PLUGIN})
    endif()
    add_dependencies(${name} engine${suffix})
    target_link_libraries(${name} PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
    if(suffix STREQUAL "_asan")
      target_compile_options(${name} PRIVATE ${ENGINE_ASAN_COMPILE_OPTIONS})
      target_link_options(${name} PRIVATE ${ENGINE_ASAN_LINK_OPTIONS})
    elseif(suffix STREQUAL "_tsan")
      target_compile_options(${name} PRIVATE ${ENGINE_TSAN_COMPILE_OPTIONS})
      target_link_options(${name} PRIVATE ${ENGINE_TSAN_LINK_OPTIONS})
    endif()
    add_test(NAME ${name} COMMAND ${name})
    if(ARG_PLANTED)
      set_tests_properties(${name} PROPERTIES WILL_FAIL TRUE)
    endif()
    if(ARG_PASS_REGEX)
      set_tests_properties(${name} PROPERTIES PASS_REGULAR_EXPRESSION "${ARG_PASS_REGEX}")
    endif()
  else()
    if(ENGINE_BUILD_SHARED)
      set(why "this configuration builds no engine${suffix} library")
    else()
      set(why "ENGINE_BUILD_SHARED is OFF")
    endif()
    add_test(NAME ${name} COMMAND ${CMAKE_COMMAND} -E echo "skipped: ${why}")
    set_tests_properties(${name} PROPERTIES SKIP_REGULAR_EXPRESSION "skipped: ")
  endif()
  set_tests_properties(${name} PROPERTIES LABELS "engine_library;${flavor_label}")
  engine_label_tests(${name})
endfunction()

# Day 346: the C API core. `host_c` is a C program linked statically against `engine_c` (plain, asan, tsan: the C API without the shared library; Day 347 loads it
# with dlopen): it creates an engine with settings and a log callback, queries the version, destroys the engine and makes every bad call the API documents.
# `engine_c_status_map_test` checks the table that maps the engine's result types to the C statuses.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND TARGET engine_c)
  include(Catch)
  foreach(suffix "" "_asan" "_tsan")
    if(NOT TARGET engine_c${suffix})
      continue()
    endif()
    set(flavor_label "")
    set(test_suffix "")
    if(suffix STREQUAL "_asan")
      set(flavor_label asan)
      set(test_suffix " [asan]")
    elseif(suffix STREQUAL "_tsan")
      set(flavor_label tsan)
      set(test_suffix " [tsan]")
    endif()
    add_executable(host_c${suffix} host_c.c)
    target_include_directories(host_c${suffix} PRIVATE ${CMAKE_SOURCE_DIR}/engine)
    target_link_libraries(host_c${suffix} PRIVATE engine_c${suffix} Threads::Threads)
    set_target_properties(host_c${suffix} PROPERTIES LINKER_LANGUAGE CXX)  # the static libraries are C++: the C program links with the C++ driver
    # Day 349: the runtime (frames, worlds, entities, components, queries, counters, jobs, the crash handler) through the table, and two host threads at once.
    add_executable(host_runtime${suffix} host_runtime.c)
    target_include_directories(host_runtime${suffix} PRIVATE ${CMAKE_SOURCE_DIR}/engine)
    target_compile_definitions(host_runtime${suffix} PRIVATE "RUNTIME_DUMP_DIR=\"${CMAKE_CURRENT_BINARY_DIR}/runtime_dump${suffix}\"")
    target_link_libraries(host_runtime${suffix} PRIVATE engine_c${suffix} Threads::Threads m)
    set_target_properties(host_runtime${suffix} PROPERTIES LINKER_LANGUAGE CXX)
    add_executable(engine_c_tests${suffix} status_map_test.cpp)
    target_link_libraries(engine_c_tests${suffix} PRIVATE engine_c${suffix} Catch2::Catch2WithMain Threads::Threads)
    if(suffix STREQUAL "_asan")
      target_compile_options(host_c_asan PRIVATE ${ENGINE_ASAN_COMPILE_OPTIONS})
      target_link_options(host_c_asan PRIVATE ${ENGINE_ASAN_LINK_OPTIONS})
      target_compile_options(host_runtime_asan PRIVATE ${ENGINE_ASAN_COMPILE_OPTIONS})
      target_link_options(host_runtime_asan PRIVATE ${ENGINE_ASAN_LINK_OPTIONS})
      engine_asan_runtime(engine_c_tests_asan)
    elseif(suffix STREQUAL "_tsan")
      target_compile_options(host_c_tsan PRIVATE ${ENGINE_TSAN_COMPILE_OPTIONS})
      target_link_options(host_c_tsan PRIVATE ${ENGINE_TSAN_LINK_OPTIONS})
      target_compile_definitions(host_runtime_tsan PRIVATE RUNTIME_NO_CRASH_CHILD=1)
      target_compile_options(host_runtime_tsan PRIVATE ${ENGINE_TSAN_COMPILE_OPTIONS})
      target_link_options(host_runtime_tsan PRIVATE ${ENGINE_TSAN_LINK_OPTIONS})
    endif()
    add_test(NAME engine_c_host${suffix} COMMAND host_c${suffix})
    set_tests_properties(engine_c_host${suffix} PROPERTIES LABELS "engine_library;${flavor_label}" PASS_REGULAR_EXPRESSION "host_c: ok")
    engine_label_tests(engine_c_host${suffix})
    add_test(NAME engine_c_runtime${suffix} COMMAND host_runtime${suffix})
    set_tests_properties(engine_c_runtime${suffix} PROPERTIES LABELS "engine_library;${flavor_label}" PASS_REGULAR_EXPRESSION "host_runtime: ok")
    engine_label_tests(engine_c_runtime${suffix})
    engine_discover_tests(engine_c_tests${suffix} TEST_SUFFIX "${test_suffix}" PROPERTIES LABELS "engine_library;${flavor_label}")
  endforeach()
endif()

# Day 348: shutdown, unload and reload. host_reload: 50 load-run-destroy-unload cycles in one process (the library gone from /proc/self/maps after each, the
# thread count back, the resident size flat). host_inside: a host thread blocked in the log callback holds destroy until it returns.
foreach(suffix "" "_asan" "_tsan")
  # Under asan the allocator's quarantine and shadow memory grow the resident size for a while: the limit is looser.
  if(suffix STREQUAL "_asan")
    set(growth_kib 65536)
  elseif(suffix STREQUAL "_tsan")
    set(growth_kib 65536)
  else()
    set(growth_kib 8192)
  endif()
  engine_library_c_test(engine_reload_cycles${suffix} "${suffix}" SOURCE host_reload.c PASS_REGEX "host_reload: ok"
                        DEFINES "ENGINE_LIB_FILE=\"$<TARGET_FILE_NAME:engine${suffix}>\"" RESIDENT_GROWTH_LIMIT_KIB=${growth_kib})
  engine_library_c_test(engine_destroy_waits_for_callback${suffix} "${suffix}" SOURCE host_inside.c PASS_REGEX "host_inside: ok")
endforeach()
